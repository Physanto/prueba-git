# Comandos Usados en la Actividad Propuesta

- git init "ruta relativa o absoluta del proyecto"
- git add "ruta relativa o absoluta de los archivos"
- git commit -m "mensaje descriptivo"
- git status
- git pull origin "rama objetivo"
- git push origin "rama a subir"
- git remote add origin "url del repositorio remoto"
- git config --global user.name "nombre"
- git config --global user.email "email"
- git config --list
- git branch
- git checkout "nombre de la rama"
- git log
- git clone "url del repositorio que se desea clonar"
